```rust
pub enum SpatialError {
    #[error("non-finite volume or coordinate")] NonFinite,
    #[error("invalid volume: {0}")] InvalidVolume(&'static str),
    #[error("invalid index configuration: {0}")] InvalidConfig(&'static str),

    #[error("entity {0:?} is already in the index")] DuplicateEntity(EntityId),
    #[error("entity {0:?} is not in the index")] UnknownEntity(EntityId),
}
```

```rust
  pub fn min(&self) -> DVec3; pub fn max(&self) -> DVec3; pub fn center(&self) -> DVec3; pub fn half_extents(&self) -> DVec3;
  pub fn intersects(&self, o: &Aabb) -> bool /* inclusive */; pub fn contains_point(&self, p: DVec3) -> bool; }
#[derive(Clone, Copy, Debug, PartialEq)] pub struct Sphere { center: DVec3, radius: f64 }
impl Sphere { pub fn new(center: DVec3, radius: f64) -> Result<Self, SpatialError>; pub fn from_world_pos(index_frame, center: WorldPos, radius) -> Result<..>;
  pub fn center(&self) -> DVec3; pub fn radius(&self) -> f64; pub fn bounds(&self) -> Aabb; pub fn intersects_aabb(&self, b: &Aabb) -> bool /* closest-point dist^2 <= r^2 */ }
#[derive(Clone, Copy, Debug, PartialEq)] pub struct Plane { pub normal: DVec3, pub d: f64 }  // inside when normal.dot(p) + d >= 0; normal unit length
#[derive(Clone, Copy, Debug, PartialEq)] pub struct Frustum { planes: [Plane; 6], corners: [DVec3; 8], bounds: Aabb }
impl Frustum {
  pub fn from_perspective(position: DVec3, rotation: DQuat, fov_y: f64, aspect: f64, near: f64, far: f64) -> Result<Self, SpatialError>;
  pub fn from_world_pos(index_frame: FrameId, camera: WorldPos, rotation: DQuat, fov_y: f64, aspect: f64, near: f64, far: f64) -> Result<Self, SpatialError>;
  pub fn planes(&self) -> &[Plane; 6]; pub fn corners(&self) -> &[DVec3; 8]; pub fn bounds(&self) -> Aabb;
  pub fn intersects_aabb(&self, b: &Aabb) -> bool;  // step 1: if !self.bounds.intersects(b) -> false; step 2 p-vertex test: for each plane pick the box vertex furthest along normal; if normal.dot(pv)+d < 0 -> false. Conservative (may report true near corners); documented. Step 1 is REQUIRED: it guarantees every match intersects bounds(), which grid backends (plan 07) rely on to find all matches inside the bounds' cell range.
}
```

index.rs:
```rust
/// Pluggable spatial index over EntityIds in one index frame (ADR-0006).
/// Query contract: `out` is cleared, then filled with matching ids sorted ascending, without duplicates.
/// Region: bounds.intersects(region). Radius: sphere.intersects_aabb(bounds). Frustum: frustum.intersects_aabb(bounds) (conservative).
pub trait SpatialIndex {
    fn index_frame(&self) -> FrameId;
    fn insert(&mut self, id: EntityId, bounds: Aabb) -> Result<(), SpatialError>;
    fn update(&mut self, id: EntityId, bounds: Aabb) -> Result<(), SpatialError>;
    fn remove(&mut self, id: EntityId) -> bool;
    fn bounds(&self, id: EntityId) -> Option<Aabb>;
    fn query_frustum(&self, frustum: &Frustum, out: &mut Vec<EntityId>);
    fn query_region(&self, region: &Aabb, out: &mut Vec<EntityId>);
    fn query_radius(&self, sphere: &Sphere, out: &mut Vec<EntityId>);
    fn len(&self) -> usize;
    fn is_empty(&self) -> bool { self.len() == 0 }
    fn clear(&mut self);
}
```
The trait must be object-safe (no generics, no Self returns) so `Box<dyn SpatialIndex>` works. Uses `&Sphere` for radius queries (Claude's discretion: validated volume instead of raw center/radius; documented).
