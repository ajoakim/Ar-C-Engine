pub const DEFAULT_KAPPA: f64 = 0.25;
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct LodModel { pub radius: f64, pub max_relief: f64, pub cells_per_edge: u32, pub kappa: f64 }
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Bound { pub center: DVec3, pub radius: f64 }
impl LodModel {
    pub fn from_terrain(t: &PlanetTerrain, kappa: f64) -> LodModel;
    pub fn cell_size(&self, level: u8) -> f64;        // radius * FRAC_PI_2 / (cells_per_edge * 2^level)
    pub fn sagitta(&self, level: u8) -> f64;          // radius * theta * theta / 8, theta = FRAC_PI_2 / (cells_per_edge * 2^level)
    pub fn geometric_error(&self, level: u8) -> f64;  // max(kappa * cell_size, sagitta)
    pub fn node_bound(&self, id: PatchId) -> Bound;   // center = center_dir * radius; radius = max corner chord distance + max_relief + sagitta(level)
}
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct View { pub eye: DVec3, pub proj_scale: f64 }
impl View {
    /// proj_scale = viewport_height_px / (2 * tan_half_fov_y). None if any input is non-finite or <= 0.
    pub fn new(eye: DVec3, viewport_height_px: u32, tan_half_fov_y: f64) -> Option<View>;
}
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct LodParams { pub tau_px: f64, pub hysteresis: f64, pub max_level: u8, pub max_leaves: usize, pub tau_growth: f64, pub max_tau_steps: u32, pub min_distance: f64 }
// Default: tau_px 2.0, hysteresis 0.2, max_level 19, max_leaves 700, tau_growth 1.5, max_tau_steps 8, min_distance 1.0
#[derive(Clone, Copy, Debug, Default, PartialEq)]
pub struct SelectStats { pub nodes_visited: u32, pub leaves: usize, pub forced_splits: u32, pub tau_eff: f64, pub tau_steps: u32 }
#[derive(Clone, Debug, PartialEq)]
pub struct Selection { pub split: BTreeSet<PatchId>, pub leaves: Vec<PatchId>, pub stats: SelectStats }
pub fn screen_space_error(model: &LodModel, view: &View, id: PatchId, min_distance: f64) -> f64;
pub fn select(model: &LodModel, params: &LodParams, view: &View, prev_split: &BTreeSet<PatchId>) -> Selection;
pub fn balance(split: &mut BTreeSet<PatchId>) -> u32;          // returns number of forced splits; only inserts
pub fn leaves_of(split: &BTreeSet<PatchId>) -> Vec<PatchId>;   // sorted ascending
pub fn is_partition(leaves: &[PatchId]) -> bool;               // integer area sum == 6 * 4^MAX_LEVEL and no leaf is an ancestor of another
pub fn max_level_jump(leaves: &[PatchId]) -> u8;               // max level difference across any shared edge
pub fn is_balanced(leaves: &[PatchId]) -> bool;                // max_level_jump <= 1
/// Canonical verification viewpoints (eye positions), deterministic order:
/// "corner{0..8}_1km", "corner{0..8}_50m", "edge{0..12}_1km", "face_center_1km", "orbit_2r".
/// Altitudes are above radius + max_relief.
pub fn reference_views(radius: f64, max_relief: f64) -> Vec<(String, DVec3)>;
