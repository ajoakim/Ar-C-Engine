#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
#[repr(u8)]
pub enum Face { PosX = 0, NegX = 1, PosY = 2, NegY = 3, PosZ = 4, NegZ = 5 }
impl Face {
    pub const ALL: [Face; 6];
    pub const fn index(self) -> u8;
    pub fn from_index(i: u8) -> Option<Face>;
    pub fn from_normal(n: [i32; 3]) -> Option<Face>;
    pub const fn basis(self) -> FaceBasis;
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct FaceBasis { pub n: [i32; 3], pub u: [i32; 3], pub v: [i32; 3] }
impl FaceBasis {
    pub fn outward(&self, e: Edge) -> [i32; 3];            // NegU -> -u, PosU -> +u, NegV -> -v, PosV -> +v
    pub fn edge_with_outward(&self, w: [i32; 3]) -> Option<Edge>;
}
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
#[repr(u8)]
pub enum Edge { NegU = 0, PosU = 1, NegV = 2, PosV = 3 }  // NegU: x = 0 side, PosU: x = K-1, NegV: y = 0, PosV: y = K-1
impl Edge {
    pub const ALL: [Edge; 4];
    pub const fn opposite(self) -> Edge;
    pub const fn is_u_edge(self) -> bool;                   // NegU | PosU (along-edge coordinate is y / j)
    /// Grid vertex (i, j) of the t-th vertex (t in 0..=n) along this edge of an n x n cell grid,
    /// ordered by increasing along-edge coordinate.
    pub const fn vertex(self, n: u32, t: u32) -> (u32, u32);
}
// patch_id.rs
pub const MAX_LEVEL: u8 = 24;
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct PatchId(u64);
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum PatchIdError { BadFace(u8), BadLevel(u8), CoordOutOfRange { level: u8, x: u32, y: u32 }, StrayBits(u64) }
// impl Display + std::error::Error by hand (no thiserror: planetary-core depends on engine-sdk only)
impl PatchId {
    pub const fn root(face: Face) -> PatchId;
    pub fn new(face: Face, level: u8, x: u32, y: u32) -> Result<PatchId, PatchIdError>;
    pub fn from_raw(raw: u64) -> Result<PatchId, PatchIdError>;
    pub const fn raw(self) -> u64;
    pub fn face(self) -> Face;
    pub const fn level(self) -> u8;
    pub fn xy(self) -> (u32, u32);
    pub fn parent(self) -> Option<PatchId>;                 // None at level 0
    pub fn children(self) -> Option<[PatchId; 4]>;          // None at MAX_LEVEL; order [(2x,2y),(2x+1,2y),(2x,2y+1),(2x+1,2y+1)]
    pub fn ancestor_at(self, level: u8) -> Option<PatchId>; // None if level > self.level()
    pub fn is_ancestor_of(self, other: PatchId) -> bool;    // strict
    pub fn level_iter(level: u8) -> impl Iterator<Item = PatchId>; // all 6*4^level ids, ascending raw order
}
// adjacency.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Neighbor { pub patch: PatchId, pub edge: Edge, pub flipped: bool }
/// Same-level neighbor across `e`; `edge` is the neighbor's edge facing back; `flipped` is true
/// when the along-edge coordinate runs in the opposite direction on the neighbor (t' = K-1-t,
/// and for vertex index on an n-cell edge: t_vertex' = n - t_vertex).
pub fn neighbor(p: PatchId, e: Edge) -> Neighbor;
/// The two children of `neighbor(p, e).patch` that touch the shared edge, ordered by increasing
/// along-edge coordinate of `p`. None when p.level() == MAX_LEVEL.
pub fn neighbor_children(p: PatchId, e: Edge) -> Option<[PatchId; 2]>;
