```rust
// effective.rs
#[derive(Clone, Debug, PartialEq)]
pub struct EffectiveCut { pub split: BTreeSet<PatchId>, pub leaves: Vec<PatchId>, pub coarsened: u32 }
/// Top-down from the 6 roots: a resident node is drawn as a leaf unless it is in `wanted_split` AND all four
/// children are resident, in which case it is split. Non-resident roots are omitted (warm-up only).
/// Then coarsen to fixpoint: while a drawn leaf p (level L) has an edge-neighbor drawn leaf at level <= L-2,
/// remove parent(p) and its whole subtree from the split set. Leaves sorted ascending.
pub fn effective_cut(wanted_split: &BTreeSet<PatchId>, is_resident: &dyn Fn(PatchId) -> bool) -> EffectiveCut;
/// Same as effective_cut without the coarsen stage (exposed for the negative-control test and diagnostics).
pub fn naive_cut(wanted_split: &BTreeSet<PatchId>, is_resident: &dyn Fn(PatchId) -> bool) -> EffectiveCut;
// requests.rs
/// For each wanted leaf, the coarsest non-resident ancestor-or-self whose parent is resident (or which is a root);
/// skips pending ids; deduplicated; sorted by (level, raw id); truncated to `max`.
pub fn plan_requests(wanted_leaves: &[PatchId], is_resident: &dyn Fn(PatchId) -> bool, is_pending: &dyn Fn(PatchId) -> bool, max: usize) -> Vec<PatchId>;
pub fn wanted_closure(wanted_leaves: &[PatchId]) -> BTreeSet<PatchId>;          // leaves + all ancestors
pub fn stale_requests(requested: &BTreeSet<PatchId>, closure: &BTreeSet<PatchId>) -> Vec<PatchId>; // requested - closure, sorted
pub fn priority_for_level(level: u8) -> Priority;                                // <= 3 -> High, else Normal
```

`effective.rs`: build top-down with an explicit stack starting from resident roots in Face::ALL order; for node p: if `wanted_split.contains(&p)` and `p.children()` is Some and all four children are resident, insert p into split and push the children; otherwise p is a leaf.
