```rust
// mesh.rs
pub const SKIRT_MARGIN: f64 = 1.5;
#[derive(Clone, Debug, PartialEq)]
pub struct PatchMesh { pub id: PatchId, pub origin: DVec3, pub vertices: Vec<VertexPosNormalUv>, pub indices: Vec<u32>, pub bound_radius: f32, pub skirt_depth: f64 }
pub fn vertex_count(n: u32) -> usize;   // (n+1)^2 grid + 4*(n+1) skirt
pub fn index_count(n: u32) -> usize;    // 6*n*n + 4*6*n
pub fn grid_indices(n: u32) -> Vec<u32>; // identical for all patches; grid triangles first, then skirts in Edge::ALL order
pub fn skirt_depth(model: &LodModel, level: u8) -> f64;  // SKIRT_MARGIN * model.geometric_error(level.saturating_sub(1))
pub fn vertex_position_f64(terrain: &PlanetTerrain, id: PatchId, i: u32, j: u32) -> DVec3;
pub fn border_positions_f64(terrain: &PlanetTerrain, id: PatchId, edge: Edge) -> Vec<DVec3>; // n+1, Edge::vertex order
pub fn build_patch(terrain: &PlanetTerrain, model: &LodModel, id: PatchId) -> PatchMesh;
pub fn build_patch_cancellable(terrain: &PlanetTerrain, model: &LodModel, id: PatchId, cancelled: &dyn Fn() -> bool) -> Option<PatchMesh>; // checks between rows
// Vertex order: grid vertex (i, j) at index j*(n+1)+i; skirt vertex t of edge e at (n+1)^2 + e as usize*(n+1) + t.
// payload.rs
pub const PAYLOAD_MAGIC: u32 = 0x4843_5450; // b"PTCH" little-endian
pub const PAYLOAD_VERSION: u32 = 1;
pub const HEADER_LEN: usize = 60; // magic u32, version u32, id u64, vertex_count u32, index_count u32, origin 3*f64, bound_radius f32, skirt_depth f64
pub fn payload_len(n: u32) -> usize;  // HEADER_LEN + vertex_count*32 + index_count*4
pub fn encode_payload(mesh: &PatchMesh) -> Vec<u8>;
#[derive(Clone, Debug, PartialEq)]
pub enum PayloadError { Truncated { expected: usize, got: usize }, BadMagic(u32), BadVersion(u32), IdMismatch { expected: u64, got: u64 }, CountMismatch { vertices: u32, indices: u32 }, NonFinite, IndexOutOfRange(u32) }
pub fn decode_payload(bytes: &[u8], expected: PatchId, n: u32) -> Result<PatchMesh, PayloadError>;
```
