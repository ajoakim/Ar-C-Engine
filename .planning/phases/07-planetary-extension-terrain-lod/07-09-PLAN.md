// descent.rs
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct DescentScript { pub start_dir: DVec3, pub start_altitude: f64, pub end_altitude: f64, pub descent_seconds: f64, pub lateral_radians: f64, pub pitch_down_radians: f64 }
impl DescentScript {
    /// Default demo path: start_dir normalize(1.0, 0.8, 0.6) (heads toward a cube corner region), start_altitude = radius,
    /// end_altitude 3.0, descent_seconds 8.0, lateral_radians 0.3, pitch_down_radians 0.5.
    pub fn orbit_to_surface(radius: f64) -> Self;
    /// Pose at simulated time t (clamped to [0, descent_seconds]): altitude decays geometrically
    /// start -> end; direction rotates by lateral_radians * (t / T) about a fixed perpendicular axis;
    /// pos = dir * (surface_radius_at(dir) + altitude(t)); orientation looks along the travel tangent pitched down.
    pub fn pose_at(&self, t: f64, surface_radius_at: &dyn Fn(DVec3) -> f64) -> (DVec3, DQuat);
}
// lib.rs
#[derive(Clone, Debug, PartialEq)]
pub struct AppSettings { pub terrain: TerrainConfig, pub camera: CameraParams, pub start_altitude: f64, pub viewport_height: u32, pub script: Option<DescentScript> }
// Default: TerrainConfig::default(), CameraParams::default(), start_altitude = TerrainParams::default().radius, viewport_height 720, script None
pub struct PlanetaryApp { /* settings, terrain: TerrainSystem, camera: PlanetaryCamera, viewport_h, time, last_view */ }
impl PlanetaryApp {
    pub fn new(settings: AppSettings) -> Result<Self, SdkError>;
    pub fn terrain(&self) -> &TerrainSystem;
    pub fn camera(&self) -> &PlanetaryCamera;
    pub fn report(&self) -> &TerrainReport;
    pub fn simulated_time(&self) -> f64;
    pub fn shutdown(self);
}
impl App for PlanetaryApp { setup, input, update, gpu_frame }
