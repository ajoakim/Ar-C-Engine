// error.rs (add)
#[derive(Debug, thiserror::Error)]
pub enum EngineError {
    #[error(transparent)] Lifecycle(#[from] LifecycleError),
    #[error(transparent)] Registry(#[from] RegistryError),
    #[error(transparent)] Config(#[from] ConfigError),
    #[error(transparent)] Diagnostics(#[from] engine_diagnostics::DiagnosticsError),
    #[error("module `{module}` failed: {source}")] Module { module: &'static str, #[source] source: ModuleError },
    #[error("modules can only be registered before initialize (phase {phase:?})")] RegistrationClosed { phase: EnginePhase },
    #[error("engine has not been initialized")] NotInitialized,
    #[error("engine is already initialized")] AlreadyInitialized,
    #[error("tick requires the Running phase, current phase is {phase:?}")] NotRunning { phase: EnginePhase },
}
// engine.rs
pub struct Engine { config: EngineConfig, lifecycle: Lifecycle, registry: ModuleRegistry, clock: engine_diagnostics::FrameClock, initialized: Vec<&'static str> /* names whose init succeeded */, init_attempted: bool }
impl Engine {
    pub fn new(config: EngineConfig) -> Self;                                  // FrameClock::new(None, config.engine.frame_history)
    pub fn from_ron_str(text: &str) -> Result<Self, EngineError>;
    pub fn from_config_file(path: &std::path::Path) -> Result<Self, EngineError>;
    pub fn config(&self) -> &EngineConfig;
    pub fn phase(&self) -> EnginePhase;
    pub fn register_module(&mut self, module: Box<dyn Module>) -> Result<(), EngineError>;
    pub fn install_diagnostics(&mut self) -> Result<(), EngineError>;          // engine_diagnostics::init + attach
    pub fn attach_frame_timing(&mut self, handle: engine_diagnostics::FrameTimingHandle);

    pub fn load(&mut self) -> Result<(), EngineError>;
    pub fn start(&mut self) -> Result<(), EngineError>;
    pub fn tick(&mut self, dt: f64) -> Result<engine_diagnostics::FrameStats, EngineError>;
    pub fn suspend(&mut self) -> Result<(), EngineError>;
    pub fn resume(&mut self) -> Result<(), EngineError>;
    pub fn shutdown(&mut self) -> Result<(), EngineError>;
    pub fn last_frame_stats(&self) -> Option<&engine_diagnostics::FrameStats>;
    pub fn frame_history(&self) -> impl Iterator<Item = &engine_diagnostics::FrameStats> + '_;
}

Borrow note: split borrows by destructuring `let Engine { config, registry, initialized, .. } = self;` before the closure.
- `load`: if `!init_attempted` or `initialized.len() != registry.len()` -> NotInitialized; `lifecycle.transition(Load)?`; modules `load()` in order (errors -> EngineError::Module).
- `start`: `lifecycle.transition(Running)?`.
- `tick(dt)`: if phase != Running -> NotRunning{phase}. `clock.begin_frame()`; inside `let frame_span = tracing::info_span!("engine_frame", dt).entered();` iterate modules in order, each inside `tracing::info_span!("module_update", module = m.name()).in_scope(|| m.update(dt))` mapping errors to EngineError::Module; `drop(frame_span)` BEFORE `clock.end_frame()?` so the engine_frame span closes inside the frame; on a module error still call `clock.end_frame()` (ignore its result) before returning the error so the next begin_frame starts clean. Return the FrameStats and `tracing::trace!(frame = stats.frame_index, ?stats.frame_time, "frame complete")`.
- `suspend`: `lifecycle.transition(Suspend)?` then `for_each_in_reverse(|m| { m.suspend(); Ok(()) })`. `resume`: if phase != Suspend return `EngineError::Lifecycle(LifecycleError::IllegalTransition{ from: phase, to: EnginePhase::Running })` (needed because Load->Running is a legal table edge but is `start`, not `resume`); otherwise `transition(Running)?` then `resume()` hooks in order.
- `shutdown`: `lifecycle.transition(Shutdown)?`; then `for_each_in_reverse` calling `m.shutdown()` only for modules whose name is in a cloned `initialized` list; `tracing::info!(app = %config.engine.app_name, "engine shut down")`.
- Log every successful phase change at `tracing::info!(phase = ?new_phase, "engine phase")` (the Lifecycle already logs at debug).
