// lifecycle.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub enum EnginePhase { Initialize, Load, Running, Suspend, Shutdown }
impl EnginePhase { pub const ALL: [EnginePhase; 5]; }
pub fn is_legal_transition(from: EnginePhase, to: EnginePhase) -> bool;
#[derive(Debug)] pub struct Lifecycle { phase: EnginePhase }
impl Lifecycle { pub fn new() -> Self /* Initialize */; pub fn phase(&self) -> EnginePhase; pub fn transition(&mut self, to: EnginePhase) -> Result<(), LifecycleError>; }
impl Default for Lifecycle
// error.rs
#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
pub enum LifecycleError { #[error("illegal lifecycle transition {from:?} -> {to:?}")] IllegalTransition { from: EnginePhase, to: EnginePhase } }
#[derive(Debug, thiserror::Error)]
pub enum ConfigError {
    #[error("failed to read config file {path}: {source}")] Io { path: std::path::PathBuf, #[source] source: std::io::Error },
    #[error("config file {path} is {size} bytes, limit is {limit}")] TooLarge { path: std::path::PathBuf, size: u64, limit: u64 },
    #[error("invalid RON config: {0}")] Parse(#[from] ron::error::SpannedError),
    #[error("invalid config section for module `{module}`: {source}")] ModuleSection { module: String, #[source] source: ron::error::SpannedError },
}
#[derive(Debug, thiserror::Error)]
pub enum ModuleError { #[error("{0}")] Message(String), #[error(transparent)] Config(#[from] ConfigError) }
#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
pub enum RegistryError {
    #[error("module `{0}` registered twice")] DuplicateModule(String),
    #[error("module `{module}` depends on unregistered module `{dependency}`")] MissingDependency { module: String, dependency: String },
    #[error("dependency cycle among modules {0:?}")] DependencyCycle(Vec<String>),
}
// config.rs
pub const MAX_CONFIG_BYTES: u64 = 1_048_576;
#[derive(Clone, Debug, PartialEq, serde::Deserialize)] #[serde(default, deny_unknown_fields)]
pub struct EngineSection { pub app_name: String /* "engine-app" */, pub log_filter: String /* "info" */, pub frame_history: usize /* 120 */ }
#[derive(Clone, Default, serde::Deserialize)] #[serde(default, deny_unknown_fields)]
pub struct EngineConfig { pub engine: EngineSection, pub modules: std::collections::BTreeMap<String, Box<ron::value::RawValue>> }
impl EngineConfig {
    pub fn from_ron_str(text: &str) -> Result<Self, ConfigError>;
    pub fn load(path: &std::path::Path) -> Result<Self, ConfigError>;   // metadata().len() > MAX_CONFIG_BYTES -> TooLarge, before reading
    pub fn module_config(&self, name: &str) -> ModuleConfig<'_>;
    pub fn module_section_names(&self) -> impl Iterator<Item = &str> + '_;
}
#[derive(Clone, Copy)] pub struct ModuleConfig<'a> { name: &'a str, section: Option<&'a ron::value::RawValue> }
impl<'a> ModuleConfig<'a> {
    pub fn empty(name: &'a str) -> Self;
    pub fn name(&self) -> &'a str; pub fn is_present(&self) -> bool;
    pub fn parse<T: serde::de::DeserializeOwned>(&self) -> Result<Option<T>, ConfigError>;          // None if section absent
    pub fn parse_or_default<T: serde::de::DeserializeOwned + Default>(&self) -> Result<T, ConfigError>;
}
// module.rs
pub trait Module: Send {
    fn name(&self) -> &'static str;
    fn dependencies(&self) -> &[&'static str] { &[] }
    fn init(&mut self, _config: &ModuleConfig<'_>) -> Result<(), ModuleError> { Ok(()) }
    fn load(&mut self) -> Result<(), ModuleError> { Ok(()) }
    fn update(&mut self, _dt: f64) -> Result<(), ModuleError> { Ok(()) }
    fn suspend(&mut self) {}
    fn resume(&mut self) {}
    fn shutdown(&mut self) {}
}
// registry.rs
#[derive(Default)] pub struct ModuleRegistry { /* modules: Vec<Box<dyn Module>>, order: Option<Vec<usize>> */ }
impl ModuleRegistry {
    pub fn new() -> Self;
    pub fn register(&mut self, module: Box<dyn Module>) -> Result<(), RegistryError>;   // DuplicateModule by name(); invalidates cached order
    pub fn len(&self) -> usize; pub fn is_empty(&self) -> bool;
    pub fn resolve_order(&mut self) -> Result<Vec<&'static str>, RegistryError>;     // caches order, returns names in init order
    pub fn for_each_in_order<E: From<RegistryError>>(&mut self, f: impl FnMut(&mut dyn Module) -> Result<(), E>) -> Result<(), E>;  // resolves order if needed; dependency order
    pub fn for_each_in_reverse<E: From<RegistryError>>(&mut self, f: impl FnMut(&mut dyn Module) -> Result<(), E>) -> Result<(), E>; // resolves order if needed; reverse dependency order
}
// lib.rs: private modules, re-export all public items above.
