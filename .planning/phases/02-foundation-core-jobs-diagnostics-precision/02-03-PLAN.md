#[derive(Debug, thiserror::Error)]
pub enum DiagnosticsError {
    #[error("a global tracing subscriber is already installed")] AlreadyInitialized,
    #[error("invalid log filter `{filter}`: {reason}")] InvalidFilter { filter: String, reason: String },
    #[error("end_frame called without a matching begin_frame")] FrameNotStarted,
}
// layer.rs
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpanTiming { pub name: &'static str, pub total: std::time::Duration, pub count: u32 }
#[derive(Clone, Default)]
pub struct FrameTimingHandle { /* Arc<Mutex<BTreeMap<&'static str, (Duration, u32)>>> */ }
impl FrameTimingHandle {
    pub fn snapshot(&self) -> Vec<SpanTiming>;   // sorted by name, does not reset
    pub fn take(&self) -> Vec<SpanTiming>;       // sorted by name, resets accumulator
}
pub struct FrameTimingLayer { /* handle */ }
impl FrameTimingLayer { pub fn new() -> (Self, FrameTimingHandle); }
/// Registry + FrameTimingLayer only (no fmt, no filter). For scoped use with tracing::subscriber::with_default
/// in tests and headless tools, so downstream crates need no tracing-subscriber dependency.
pub fn frame_timing_subscriber() -> (impl tracing::Subscriber + Send + Sync + 'static, FrameTimingHandle);
impl<S> tracing_subscriber::Layer<S> for FrameTimingLayer where S: tracing::Subscriber + for<'a> tracing_subscriber::registry::LookupSpan<'a>
// frame_stats.rs
#[derive(Clone, Debug, PartialEq)]
pub struct FrameStats { pub frame_index: u64, pub frame_time: std::time::Duration, pub spans: Vec<SpanTiming> }
pub struct FrameClock { /* timing: Option<FrameTimingHandle>, started: Option<Instant>, next_index: u64, history: VecDeque<FrameStats>, capacity: usize */ }
impl FrameClock {
    pub fn new(timing: Option<FrameTimingHandle>, history_capacity: usize) -> Self; // capacity clamped to >= 1
    pub fn set_timing(&mut self, timing: Option<FrameTimingHandle>);
    pub fn begin_frame(&mut self);                                   // records Instant::now(), discards spans recorded before frame start via take()
    pub fn end_frame(&mut self) -> Result<FrameStats, DiagnosticsError>; // FrameNotStarted if no begin
    pub fn last(&self) -> Option<&FrameStats>;
    pub fn history(&self) -> impl Iterator<Item = &FrameStats> + '_;   // oldest first, len <= capacity
}
// init.rs
#[derive(Clone, Debug)]
pub struct DiagnosticsConfig { pub filter: String, pub ansi: bool }   // Default: filter "info", ansi true
pub fn init(config: &DiagnosticsConfig) -> Result<FrameTimingHandle, DiagnosticsError>;

Layer implementation (tracing-subscriber 0.3 `Layer` trait):
- `on_new_span(&self, _attrs, id, ctx)`: `if let Some(span) = ctx.span(id) { span.extensions_mut().insert(SpanStart(Instant::now())); }` where `struct SpanStart(Instant);`.
- `on_close(&self, id, ctx)`: look up the span, read `SpanStart` from extensions, compute `elapsed()`, and add to the accumulator entry for `span.metadata().name()` (`&'static str`), incrementing count. Timing semantic is creation-to-close (research Pitfall 6, chosen for simplicity); document this on `SpanTiming`.

pub fn init(config: &DiagnosticsConfig) -> Result<FrameTimingHandle, DiagnosticsError> {
