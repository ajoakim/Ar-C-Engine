pub enum Priority { Background, Low, Normal, High, Critical }

impl CancelToken { pub fn new() -> Self; pub fn cancel(&self); pub fn is_cancelled(&self) -> bool; } // Release store / Acquire load
// error.rs
#[derive(Debug, thiserror::Error)]
pub enum JobError {
    #[error("job system needs at least one worker")] ZeroWorkers,
    #[error("failed to spawn worker thread: {0}")] Spawn(#[from] std::io::Error),
}
// scheduler.rs
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum JobStatus { Completed, Cancelled, Panicked }
pub struct JobHandle { /* token: CancelToken, done: crossbeam_channel::Receiver<JobStatus>, status cache */ }
impl JobHandle {
    pub fn cancel(&self); pub fn is_cancelled(&self) -> bool; pub fn token(&self) -> CancelToken;
    pub fn try_status(&self) -> Option<JobStatus>;   // non-blocking
    pub fn wait(self) -> JobStatus;                  // blocks; returns Cancelled if the job was dropped without reporting
}
pub struct JobSystemBuilder { workers: usize, start_paused: bool, thread_name: String }
impl JobSystemBuilder {
    pub fn workers(self, n: usize) -> Self; pub fn start_paused(self, paused: bool) -> Self; pub fn thread_name(self, prefix: &str) -> Self;
    pub fn build(self) -> Result<JobSystem, JobError>;
}
pub struct JobSystem { /* Arc<Shared{ queue: Mutex<State{ heap: BinaryHeap<Entry>, paused: bool, shutting_down: bool, next_seq: u64 }>, cv: Condvar }>, workers: Vec<JoinHandle<()>> */ }
impl JobSystem {
    pub fn builder() -> JobSystemBuilder;            // defaults: workers = max(1, available_parallelism - 1), start_paused false, thread_name "engine-job"
    pub fn submit<F>(&self, priority: Priority, job: F) -> JobHandle where F: FnOnce(&CancelToken) + Send + 'static;
    pub fn pause(&self); pub fn resume(&self);
    pub fn queued_len(&self) -> usize;
    pub fn worker_count(&self) -> usize;
    pub fn shutdown(self);                           // resumes if paused, drains the queue (cancelled entries resolve Cancelled), joins all workers
}

Queue design (research Pitfall 1: crossbeam channels are FIFO, so priority uses a heap under a lock; channels are only for results):
- `struct Entry { priority: Priority, seq: u64, token: CancelToken, job: Box<dyn FnOnce(&CancelToken) + Send>, done: crossbeam_channel::Sender<JobStatus> }`; implement `Ord`/`PartialOrd`/`Eq`/`PartialEq` manually comparing `(priority, std::cmp::Reverse(seq))` only.
- `submit`: create `CancelToken::new()` and `crossbeam_channel::bounded(1)`; lock state, push Entry with `seq = next_seq` (then increment), `cv.notify_one()`, return JobHandle.
- Worker loop (thread named `format!("{prefix}-{i}")` via `std::thread::Builder::new().name(..).spawn(..)?` -> JobError::Spawn): lock; `while (state.paused && !state.shutting_down) || (state.heap.is_empty() && !state.shutting_down) { state = cv.wait(state) }`; if heap empty and shutting_down -> exit; pop Entry; unlock. If `entry.token.is_cancelled()` -> send Cancelled, drop the closure without running (research Pitfall 3). Else run inside `tracing::trace_span!("job", priority = ?entry.priority)` with `std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| (entry.job)(&entry.token)))`: Err -> Panicked; Ok -> if token cancelled then Cancelled else Completed. Ignore send errors (`let _ = done.send(..)`: the handle may be dropped).
- `pause`/`resume` set `paused` and `notify_all`. `shutdown(self)`: set `paused = false`, `shutting_down = true`, `notify_all`, join every worker (workers drain the remaining heap before exiting). Implement shutdown by moving the join logic into a private `fn stop(&mut self, cancel_queued: bool)` used by both `shutdown` (cancel_queued false) and `Drop` (cancel_queued true: call `token.cancel()` on every queued entry first); `stop` is idempotent (workers Vec drained via `std::mem::take`).
- `JobHandle::wait(self)`: `self.done.recv().unwrap_or(JobStatus::Cancelled)` (sender dropped without report counts as Cancelled; never hang). `try_status` uses `try_recv` and caches the received status in a `std::sync::OnceLock<JobStatus>` field so repeated calls return the same value; `wait` returns the cached value first if present, otherwise blocks on `recv`.
- Lock poisoning: `lock().unwrap_or_else(PoisonError::into_inner)`; condvar wait likewise. No `unwrap`/`expect`/`panic!` in non-test code.
Unit tests in each module's `#[cfg(test)] mod tests` for the behavior list.
