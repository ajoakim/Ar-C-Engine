```rust
// error.rs
#[derive(Debug, thiserror::Error)] #[non_exhaustive]
pub enum SdkError { Engine(#[from] EngineError), Render(#[from] RenderError), Platform(#[from] PlatformError), Math(#[from] MathError), App(String) }
// app.rs
#[derive(Debug, Clone, Copy, PartialEq, Eq)] pub enum AppControl { Continue, Exit }
pub struct SceneState { pub frames: FrameRegistry, pub camera: Camera, pub items: Vec<RenderItem>, pub clear_color: [f32; 4] }
impl SceneState { pub fn new() -> Self; } // camera at root origin, clear [0.02, 0.02, 0.05, 1.0]
pub struct Resources<'a> { /* backend: &'a mut dyn RenderBackend */ }
impl Resources<'_> { pub fn create_mesh(&mut self, label: &'static str, vertices: &[VertexPosColor]) -> Result<MeshHandle, SdkError>; }
pub trait App {
    fn setup(&mut self, resources: &mut Resources<'_>, scene: &mut SceneState) -> Result<(), SdkError>;
    fn input(&mut self, _event: &InputEvent, _scene: &mut SceneState) {}
    fn update(&mut self, dt: f64, scene: &mut SceneState) -> AppControl;
}
// runtime.rs
pub struct Runtime<B: RenderBackend> { /* engine, backend, scene, compiled: CompiledGraph, pipeline: PipelineId, frame_index: u64 */ }
impl<B: RenderBackend> Runtime<B> {
    pub fn new(engine: Engine, backend: B) -> Result<Self, SdkError>;
    pub fn setup(&mut self, app: &mut dyn App) -> Result<(), SdkError>;
    pub fn handle_input(&mut self, app: &mut dyn App, event: &InputEvent);
    pub fn frame(&mut self, app: &mut dyn App, dt: f64) -> Result<AppControl, SdkError>;
    pub fn shutdown(&mut self) -> Result<(), SdkError>;
    pub fn backend(&self) -> &B; pub fn engine(&self) -> &Engine; pub fn scene(&self) -> &SceneState;
}
// run.rs
#[derive(Debug, Clone, Default)] pub struct AppConfig { pub window: WindowConfig, pub engine: EngineConfig }
pub fn run_app<A: App>(app: A, config: AppConfig) -> Result<(), SdkError>;
```
