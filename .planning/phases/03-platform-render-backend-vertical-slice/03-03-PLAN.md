```rust
// input.rs
#[non_exhaustive] #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum KeyCode { W, A, S, D, Q, E, Up, Down, Left, Right, Space, Escape, Enter, ShiftLeft, ControlLeft, Other }
#[non_exhaustive] #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum MouseButton { Left, Right, Middle, Back, Forward, Other(u16) }
#[non_exhaustive] #[derive(Debug, Clone, Copy, PartialEq)]
pub enum InputEvent {
    KeyPressed { key: KeyCode, repeat: bool }, KeyReleased { key: KeyCode },
    MouseButtonPressed(MouseButton), MouseButtonReleased(MouseButton),
    CursorMoved { x: f64, y: f64 }, MouseWheel { lines_x: f32, lines_y: f32 },
    Resized { width: u32, height: u32 }, Focused(bool), CloseRequested,
}
// window.rs
#[derive(Clone)] pub struct PlatformWindow { /* inner: Arc<winit::window::Window> (private) */ }
impl PlatformWindow { pub fn inner_size(&self) -> (u32, u32); pub fn request_redraw(&self); pub fn set_title(&self, title: &str); }
impl HasWindowHandle for PlatformWindow; impl HasDisplayHandle for PlatformWindow; impl std::fmt::Debug for PlatformWindow;
// event_loop.rs
#[derive(Debug, Clone, PartialEq, Eq)] pub struct WindowConfig { pub title: String, pub width: u32, pub height: u32 } // Default: "engine", 1280, 720
impl WindowConfig { pub const MAX_DIMENSION: u32 = 16384; pub const MAX_TITLE_LEN: usize = 256; pub fn validate(&self) -> Result<(), PlatformError>; }
#[derive(Debug, Clone, Copy, PartialEq, Eq)] pub enum LoopControl { Continue, Exit }
pub type HandlerError = Box<dyn std::error::Error + Send + Sync + 'static>;
pub trait PlatformHandler {
    fn window_ready(&mut self, window: PlatformWindow) -> Result<(), HandlerError>;
    fn input(&mut self, event: InputEvent);
    fn redraw(&mut self, dt: f64) -> LoopControl;
    fn exiting(&mut self) {}
}
pub fn run(config: WindowConfig, handler: &mut dyn PlatformHandler) -> Result<(), PlatformError>;
// error.rs
#[derive(Debug, thiserror::Error)] #[non_exhaustive]
pub enum PlatformError { InvalidConfig(&'static str), EventLoop(String), CreateWindow(String), Handler(HandlerError) }
```
