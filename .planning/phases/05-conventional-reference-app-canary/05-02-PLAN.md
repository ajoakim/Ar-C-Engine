```rust
//! PLAN-01 canary: no planetary crate may be reachable from the conventional app.
pub const CANARY_RULE_ID: &str = "canary-planetary-reachable";
pub const DEFAULT_CANARY_PACKAGE: &str = "conventional-3d";

#[derive(Debug, Clone, Default)]
pub struct CanaryNode { pub planetary: bool, pub deps: Vec<String> }
#[derive(Debug, Clone, Default)]
pub struct DepGraph { pub nodes: std::collections::BTreeMap<String, CanaryNode> }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CanaryViolation { pub root: String, pub reached: String, pub path: Vec<String> }
// Display: "error[canary-planetary-reachable]: {root} -> {reached}: path {path joined with " -> "}"

pub fn is_planetary(name: &str, rel_manifest_dir: &std::path::Path) -> bool;
// true if name starts with "planetary-" OR the first two Normal path components are "extensions" and "planetary"
pub fn closure(graph: &DepGraph, root: &str) -> Result<std::collections::BTreeSet<String>, CheckError>;
// BFS over deps (VecDeque + visited set), includes root; Err(CheckError::Path(format!("package `{root}` not found in dependency graph"))) if absent
pub fn reachable_planetary(graph: &DepGraph, root: &str) -> Result<Vec<CanaryViolation>, CheckError>;
// BFS with parent map; every reached node (including root) with planetary=true yields a violation whose path is root..=node reconstructed from the parent map; sorted by `reached`
pub fn canary_graph_from_manifest(manifest_path: &std::path::Path) -> Result<DepGraph, CheckError>;
pub fn check_canary(manifest_path: &std::path::Path, package: &str) -> Result<Vec<CanaryViolation>, CheckError>;
```

Build `HashMap<&PackageId, &Package>` from `md.packages`. For every `node` in `resolve.nodes`: name = `pkg.name.to_string()`; rel dir = `pkg.manifest_path.parent()` stripped of `md.workspace_root` (fallback: the absolute dir — third-party crates in the registry are then never under extensions/planetary); deps = `node.deps.iter().map(|d| packages[&d.pkg].name.to_string())` over ALL dep kinds (normal, dev, build: conservative over-approximation, documented in a doc comment). Packages sharing a name (two versions of a third-party crate) merge into one node: extend deps, OR the planetary flag (documented).
