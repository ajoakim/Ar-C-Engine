```rust
type Message = (LoadTicket, PathBuf, Result<ImportedModel, ImportError>);
pub struct ModelLoader { jobs: JobSystem, tx: mpsc::Sender<Message>, rx: mpsc::Receiver<Message>, next: u64, pending: BTreeMap<LoadTicket, (PathBuf, JobHandle)>, ready: Vec<LoadResult> }
impl ModelLoader {
    pub fn new(workers: usize) -> Result<Self, ImportError>;   // JobSystem::builder().workers(workers.max(1)).thread_name("model-import").build() map_err -> JobFailed(e.to_string())
    pub fn load(&mut self, path: impl Into<PathBuf>, limits: ImportLimits) -> LoadTicket;   // submit(Priority::Normal, move |tok| { if tok.is_cancelled() { return; } let r = import_gltf_path_with(&p, &limits); let _ = tx.send((ticket, p, r)); })
    pub fn poll(&mut self) -> Vec<LoadResult>;   // non-blocking: snapshot try_status for pending, drain rx (match by ticket, remove from pending), then any pending whose snapshot was Panicked/Cancelled -> Err(JobFailed("import job panicked"/"import job cancelled")); plus previously buffered `ready`; sorted by ticket
    pub fn wait(&mut self, ticket: LoadTicket) -> Result<ImportedModel, ImportError>;   // blocking: remove pending entry (unknown ticket -> JobFailed("unknown load ticket")), handle.wait(), then drain rx; matching ticket -> its result; other tickets' messages -> pushed to `ready`; status Panicked/Cancelled without message -> JobFailed
    pub fn wait_all(&mut self) -> Vec<LoadResult>;   // wait every pending handle, drain, map missing messages by status as above, include `ready`, sorted by ticket
    pub fn pending_len(&self) -> usize;
}
pub struct RegisteredModel { pub meshes: Vec<Handle<ImportedMesh>>, pub materials: Vec<Handle<ImportedMaterial>>, pub textures: Vec<Handle<ImportedTexture>>, pub nodes: Vec<ImportedNode> }
```

```rust
#[derive(Debug, Clone)]
pub struct FlyCamera { pub yaw: f32, pub pitch: f32, pub speed: f64, pub boost: f64, pub sensitivity: f32, held: u16, looking: bool, last_cursor: Option<(f64, f64)> }
impl Default for FlyCamera { yaw 0, pitch 0, speed 5.0 (m/s), boost 10.0, sensitivity 0.003 (rad/px), held 0, looking false, last_cursor None }
impl FlyCamera {
    pub const MAX_PITCH: f32 = 1.553_343;   // 89 degrees in radians
    pub fn new() -> Self;
    pub fn handle_input(&mut self, event: &InputEvent);
    pub fn update(&mut self, dt: f64, camera: &mut Camera) -> Result<(), SdkError>;
    pub fn orientation(&self) -> Quat;   // Quat::from_rotation_y(yaw) * Quat::from_rotation_x(pitch)
}
```
Held-key bits (private consts): FORWARD (W, Up), BACK (S, Down), LEFT (A, Left), RIGHT (D, Right), UP (E, Space), DOWN (Q, ControlLeft), BOOST (ShiftLeft). `KeyPressed` sets bits (repeat ignored effectively, idempotent), `KeyReleased` clears; `MouseButtonPressed(MouseButton::Right)` -> looking = true, last_cursor = None; `MouseButtonReleased(Right)` -> looking = false, last_cursor = None; `CursorMoved{x,y}`: if !x.is_finite() || !y.is_finite() ignore; if looking { if let Some((lx, ly)) = last_cursor { yaw -= (x - lx) as f32 * sensitivity; pitch = (pitch - (y - ly) as f32 * sensitivity).clamp(-MAX_PITCH, MAX_PITCH); } last_cursor = Some((x, y)); }; `Focused(false)` -> held = 0, looking = false, last_cursor = None; all other events (including `_` for non_exhaustive KeyCode) ignored.
`update`: if `!(dt.is_finite() && dt > 0.0)` -> just apply orientation and return Ok. Local direction `Vec3` from bits (forward = -Z, right = +X, up = +Y); rotate horizontal components by `orientation()` (vertical stays world +Y); if length > 0 normalize; `let v = dir.as_dvec3() * self.speed * if boost {self.boost} else {1.0} * dt;` `camera.position = camera.position.translated(v)?;` and `camera.orientation = self.orientation();`. Keep motion in f64 (`DVec3`) all the way to `WorldPos`.
`lib.rs`: `pub mod fly_camera;` and `pub use fly_camera::FlyCamera;`.

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-05-33 | Denial of Service | Unbounded worker threads | mitigate | Worker count fixed at construction (min 1); one JobSystem per loader |
