#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]

pub struct MaterialTable { materials: Vec<Material> }   // index 0 = Material::default(); ids never reused
impl MaterialTable { pub fn new() -> Self; pub fn add(&mut self, m: Material) -> Result<MaterialId, RenderError>; pub fn get(&self, id: MaterialId) -> Result<&Material, RenderError>; pub fn len(&self) -> usize; pub fn is_empty(&self) -> bool /* always false */ }

#[derive(Debug, Clone, Copy, PartialEq)]
