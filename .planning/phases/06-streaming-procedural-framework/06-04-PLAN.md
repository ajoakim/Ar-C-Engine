```rust
/// Named f64 parameters; ordered (BTreeMap) so iteration and equality are deterministic.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct Params { values: BTreeMap<String, f64> }
impl Params { pub fn new() -> Self; #[must_use] pub fn with(mut self, name: &str, value: f64) -> Self; pub fn get(&self, name: &str) -> Option<f64>; pub fn get_or(&self, name: &str, default: f64) -> f64; }

/// A deterministic generator layer: output = f(id, seed, params). Must be pure:
/// no globals, clocks, thread-locals, RNG state or I/O; only ADR-0008 float ops.
pub trait Generator: Send + Sync {
    /// Seed input; unique within a stack.
    fn name(&self) -> &'static str;
    /// Bump to deliberately change this layer's output (re-pin goldens).
    fn version(&self) -> u32;
    /// Number of f64 values written per id (>= 1).
    fn output_len(&self) -> usize;
    /// Write exactly `out.len() == output_len()` values.
    fn generate(&self, id: StableId, seed: u64, params: &Params, out: &mut [f64]);
}

pub struct LayerStack { world_seed: u64, layers: Vec<Box<dyn Generator>> }
impl LayerStack {
    pub fn new(world_seed: u64) -> Self;
    pub fn push(&mut self, layer: Box<dyn Generator>) -> Result<(), ProcError>;   // DuplicateLayer / EmptyLayer
    pub fn with_layer(mut self, layer: Box<dyn Generator>) -> Result<Self, ProcError>;
    pub fn world_seed(&self) -> u64;
    pub fn layer_names(&self) -> Vec<&'static str>;
    pub fn output_len(&self) -> usize;                                          // sum of layer output_len
    pub fn generate(&self, id: StableId, params: &Params) -> LayerOutput;       // for layer in &self.layers: seed = layer_seed(self.world_seed, layer.name(), layer.version()); out = vec![0.0; layer.output_len()]; layer.generate(id, seed, params, &mut out)
}
#[derive(Clone, Debug, PartialEq)]
pub struct LayerOutput { layers: Vec<(&'static str, Vec<f64>)> }
impl LayerOutput {
    pub fn layer(&self, name: &str) -> Option<&[f64]>;
    pub fn layers(&self) -> &[(&'static str, Vec<f64>)];
    /// FNV-1a-64 over, per layer in declared order: name length (u64 LE), name bytes, value count (u64 LE), canonical bits (u64 LE) of each value.
    pub fn content_hash(&self) -> Result<u64, ProcError>;
    /// Canonical little-endian bytes of all values in declared layer order (8 bytes each).
    pub fn to_le_bytes(&self) -> Result<Vec<u8>, ProcError>;
}
/// FNV-1a-64 over count (u64 LE) + canonical bits of `values`; `layer` labels errors.
pub fn hash_values(layer: &'static str, values: &[f64]) -> Result<u64, ProcError>;
```
