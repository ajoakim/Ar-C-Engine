```rust
pub const FNV_OFFSET_BASIS: u64 = 0xcbf2_9ce4_8422_2325;
pub const FNV_PRIME: u64 = 0x0000_0100_0000_01b3;
pub const fn splitmix64(mut z: u64) -> u64 {
    z = z.wrapping_add(0x9E37_79B9_7F4A_7C15);
    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}
pub const fn fnv1a64(bytes: &[u8]) -> u64 { /* while-loop over bytes: h ^= b as u64; h = h.wrapping_mul(FNV_PRIME) starting at FNV_OFFSET_BASIS */ }
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Fnv1a64 { state: u64 }
impl Fnv1a64 {
    pub const fn new() -> Self;               // state = FNV_OFFSET_BASIS
    pub fn write(&mut self, bytes: &[u8]);
    pub fn write_u32(&mut self, v: u32);      // v.to_le_bytes()
    pub fn write_u64(&mut self, v: u64);      // v.to_le_bytes()
    pub const fn finish(&self) -> u64;
}
impl Default for Fnv1a64 { fn default() -> Self { Self::new() } }   // NOT derived: derived default would be state 0
pub const fn hash2(seed: u64, ix: i64, iy: i64) -> u64 {
    splitmix64(seed ^ splitmix64((ix as u64) ^ splitmix64(iy as u64)))
}
pub const fn hash3(seed: u64, ix: i64, iy: i64, iz: i64) -> u64 {
    splitmix64(seed ^ splitmix64((ix as u64) ^ splitmix64((iy as u64) ^ splitmix64(iz as u64))))
}
/// Top 53 bits scaled by 2^-53: exact, in [0, 1).
pub fn unit_f64(h: u64) -> f64 { ((h >> 11) as f64) * (1.0 / 9_007_199_254_740_992.0) }
/// Canonical bit pattern for hashing: -0.0 -> 0.0; None for NaN or infinite.
pub fn canonical_bits(v: f64) -> Option<u64> { if !v.is_finite() { return None; } if v == 0.0 { return Some(0.0f64.to_bits()); } Some(v.to_bits()) }
```
