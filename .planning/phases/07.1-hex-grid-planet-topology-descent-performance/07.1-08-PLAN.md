```rust
pub trait Topology: Send + Sync + 'static {
    type Cell: Copy + Ord + Eq + core::fmt::Debug + Send + Sync + 'static;
    fn name(&self) -> &'static str;
    fn roots(&self) -> &[Self::Cell];
    fn level(&self, c: Self::Cell) -> u8;
    fn raw(&self, c: Self::Cell) -> u64;
    fn from_raw(&self, raw: u64) -> Option<Self::Cell>;
    fn children(&self, c: Self::Cell, out: &mut Vec<Self::Cell>);      // 4 (quad), 7 (hex), 6 (pentagon)
    fn parents(&self, c: Self::Cell, out: &mut Vec<Self::Cell>);       // 0, 1 or 2, ascending raw
    fn primary_parent(&self, c: Self::Cell) -> Option<Self::Cell>;     // lowest raw parent
    fn neighbors(&self, c: Self::Cell, out: &mut Vec<Self::Cell>);
    fn required_splits(&self, c: Self::Cell, out: &mut Vec<Self::Cell>); // cells that must be split if c is split; all strictly shallower
    fn footprint(&self, c: Self::Cell) -> Footprint;
    fn locate(&self, dir: DVec3, level: u8) -> Self::Cell;
    fn sample_angle0(&self) -> f64;                                     // sample spacing angle at level 0 for this N
    fn pieces(&self, split: &BTreeSet<Self::Cell>, out: &mut Vec<Piece<Self::Cell>>);
    fn index_variants(&self) -> Vec<IndexVariant>;
    fn build_mesh(&self, c: Self::Cell, terrain: &PlanetTerrain, model: &LodModel, debug_cells: bool, cancelled: &dyn Fn() -> bool) -> Option<CellMesh>;
}
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)] pub struct Piece<C> { pub cell: C, pub variant: u16 }
#[derive(Clone, Debug, PartialEq)] pub struct IndexVariant { pub name: &'static str, pub vertex_class: u8, pub indices: Vec<u32> }
#[derive(Clone, Debug, PartialEq)] pub struct CellMesh { pub raw: u64, pub vertex_class: u8, pub origin: DVec3, pub vertices: Vec<VertexPosNormalUv>, pub bound_radius: f32, pub skirt_depth: f64, pub height_min: f64, pub height_max: f64 }
```
